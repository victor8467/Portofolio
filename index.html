  • Automatic IP frequency threshold alerts

import re

def parse_ssh_failures(log_path):
    pattern = r"Failed password for .* from (\d+\.\d+\.\d+\.\d+)"
    attempts = {}
    with open(log_path, 'r') as f:
        for line in f:
            match = re.search(pattern, line)
            if match:
                ip = match.group(1)
                attempts[ip] = attempts.get(ip, 0) + 1
    return {ip: count for ip, count in attempts.items() if count > 5}

// C Vigenere Cipher Logic
void encrypt_vigenere(char* text, const char* key) {
    int textLen = strlen(text);
    int keyLen = strlen(key);
    for (int i = 0, j = 0; i < textLen; ++i) {
        if (isalpha(text[i])) {
            char base = isupper(text[i]) ? 'A' : 'a';
            char kbase = isupper(key[j % keyLen]) ? 'A' : 'a';
            text[i] = (text[i] - base + (key[j % keyLen] - kbase)) % 26 + base;
            j++;
        }
    }
}
